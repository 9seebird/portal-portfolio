# ── 새 앱 붙일 때 쓰는 본보기 ────────────────────────────────
#
# 손으로 만들 필요는 없다.  .\new-app.ps1 서비스ID  가 이 파일을 채워서
# apps/서비스ID.conf 를 만들어 준다.
#
# 확장자가 .conf 가 아니라 .conf.txt 인 이유:
# include /etc/nginx/apps/*.conf 에 걸리지 않게 하려는 것이다.
# 본보기가 실제 설정으로 읽히면 nginx 가 뜨지 않는다.
#
# 손으로 만든다면 아래 두 자리만 바꾸면 된다.
#   __ID__         서비스 ID   (포털 관리자 화면의 ID 와 같아야 한다)
#   __CONTAINER__  컨테이너 이름 (docker-compose 의 container_name)
#   __PORT__       컨테이너 **안쪽** 포트. 표준 규칙 v2 앱은 8080.
#                  nginx 로 화면을 주는 앱이면 80 인 경우가 많다.
#
# 서비스 ID 에 하이픈이 들어가면(meeting-room) nginx 변수 이름에는 쓸 수 없다.
# 그래서 변수 이름은 __VAR__ (하이픈을 밑줄로 바꾼 것)을 쓴다.
# ─────────────────────────────────────────────────────────────

# 직원이 주소창에 /__ID__ 를 슬래시 없이 치면 아래 location 이 안 걸려
# 포털로 넘어가 404 가 난다. 슬래시를 붙여서 돌려보낸다.
location = /__ID__ { return 301 /__ID__/; }

location /__ID__/ {
    # 먼저 포털에 "이 사람이 이걸 써도 되나"를 묻는다.
    # 401/403 이면 여기서 끝나고 앱까지 요청이 가지 않는다.
    auth_request /_authz___ID__;

    # 포털이 응답 헤더로 돌려준 사용자 정보를 받아 둔다.
    auth_request_set $u_id   $upstream_http_x_user_id;
    auth_request_set $u_name $upstream_http_x_user_name;
    auth_request_set $u_dept $upstream_http_x_user_dept;
    auth_request_set $u_role $upstream_http_x_user_role;

    set $__VAR___upstream http://__CONTAINER__:__PORT__;   # set 이 rewrite 보다 먼저
    # proxy_pass 에 변수를 쓰면 nginx 가 경로 앞부분을 잘라 주지 않는다.
    # rewrite 로 직접 잘라 주지 않으면 API 요청에 화면 HTML 이 돌아온다.
    rewrite ^/__ID__/?(.*)$ /$1 break;
    proxy_pass $__VAR___upstream;

    # 앱에는 이 헤더만 넘어간다. 앱은 이것만 읽으면 된다.
    # 이름·부서는 한글이라 퍼센트 인코딩되어 있다 (unquote 필요).
    proxy_set_header X-User-Id   $u_id;
    proxy_set_header X-User-Name $u_name;
    proxy_set_header X-User-Dept $u_dept;
    proxy_set_header X-User-Role $u_role;

    proxy_set_header Host              $host;
    proxy_set_header X-Real-IP         $remote_addr;
    proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-Prefix /__ID__;

    # 배포해도 직원 브라우저에 옛 화면이 남지 않게 한다.
    add_header Cache-Control "no-store, must-revalidate" always;
}

location = /_authz___ID__ {
    internal;                       # 브라우저가 직접 부를 수 없다
    set $authz___VAR__ http://demo-portal:8080/api/authz?service=__ID__;
    proxy_pass $authz___VAR__;
    proxy_pass_request_body off;    # 본문은 필요 없다
    proxy_set_header Content-Length "";
    proxy_set_header Host            $host;
    proxy_set_header X-Real-IP       $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Original-URI  $request_uri;
}
